    66→export default function StationsPage() {
    67→  const [stations, setStations] = React.useState<FloorStation[] | null>(null);
    68→  const [startFor, setStartFor] = React.useState<FloorStation | null>(null);
    69→  const [billFor, setBillFor] = React.useState<FloorStation | null>(null);
    70→  const [busyId, setBusyId] = React.useState<string | null>(null);
    71→
    72→  const load = React.useCallback(async () => {
    73→    try {
    74→      const res = await fetch("/api/stations", { cache: "no-store" });
    75→      if (!res.ok) throw new Error();
    76→      const json = await res.json();
    77→      setStations(json.stations);
    78→    } catch {
    79→      toast.error("Could not load stations");
    80→    }
    81→  }, []);
    82→
    83→  React.useEffect(() => {
    84→    load();
    85→    const t = setInterval(load, 10000);
    86→    return () => clearInterval(t);
    87→  }, [load]);
    88→
    89→  async function toggleMaintenance(station: FloorStation) {
    90→    setBusyId(station.id);
    91→    try {
    92→      const res = await fetch("/api/stations", {
    93→        method: "PATCH",
    94→        headers: { "Content-Type": "application/json" },
    95→        body: JSON.stringify({ stationId: station.id, action: "TOGGLE_MAINTENANCE" }),
    96→      });
    97→      const json = await res.json().catch(() => ({}));
    98→      if (!res.ok) throw new Error(json.error || "Failed");
    99→      toast.success(
   100→        res.status === 200 && json.station?.status === "MAINTENANCE"
   101→          ? `${station.name} marked under maintenance`
   102→          : `${station.name} is back online`
   103→      );
   104→      await load();
   105→    } catch (e) {
   106→      toast.error(e instanceof Error ? e.message : "Failed");
   107→    } finally {
   108→      setBusyId(null);
   109→    }
   110→  }
   111→
   112→  const counts = React.useMemo(() => {
   113→    const s = stations ?? [];
   114→    return {
   115→      free: s.filter((x) => x.status === "FREE").length,
   116→      busy: s.filter((x) => x.status === "BUSY").length,
   117→      maint: s.filter((x) => x.status === "MAINTENANCE").length,
   118→    };
   119→  }, [stations]);
   120→
   121→  return (
   122→    <div className="mx-auto max-w-7xl space-y-6">
   123→      <header className="flex flex-wrap items-center justify-between gap-3">
   124→        <div>
   125→          <h1 className="font-display text-xl font-bold sm:text-2xl">Stations</h1>
   126→          <p className="text-sm text-muted-foreground">Walk-ins, live sessions & billing</p>
   127→        </div>
   128→        <div className="flex items-center gap-2">
   129→          <Badge variant="outline" className="border-emerald-500/40 text-emerald-300">{counts.free} free</Badge>
   130→          <Badge variant="outline" className="border-amber-500/40 text-amber-300">{counts.busy} busy</Badge>
   131→          <Badge variant="outline" className="border-red-500/40 text-red-300">{counts.maint} maint.</Badge>
   132→          <Button variant="outline" size="icon" onClick={load} aria-label="Refresh" className="border-violet-500/30">
   133→            <RefreshCw className="h-4 w-4" />
   134→          </Button>
   135→        </div>
   136→      </header>
   137→
   138→      {!stations ? (
   139→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-3">
   140→          {Array.from({ length: 6 }).map((_, i) => (
   141→            <Skeleton key={i} className="h-44 rounded-2xl bg-white/5" />
   142→          ))}
   143→        </div>
   144→      ) : (
   145→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-3">
   146→          {stations.map((s) => (
   147→            <StationCard
   148→              key={s.id}
   149→              station={s}
   150→              busy={busyId === s.id}
   151→              onStart={() => setStartFor(s)}
   152→              onBill={() => setBillFor(s)}
   153→              onToggleMaintenance={() => toggleMaintenance(s)}
   154→            />
   155→          ))}
   156→        </div>
   157→      )}
   158→
   159→      {/* Start session dialog */}
   160→      <StartSessionDialog
   161→        station={startFor}
   162→        onClose={() => setStartFor(null)}
   163→        onStarted={() => {
   164→          setStartFor(null);
   165→          load();
   166→        }}
   167→      />
   168→
   169→      {/* Billing dialog */}
   170→      <BillDialog
   171→        station={billFor}
   172→        onClose={() => setBillFor(null)}
   173→        onEnded={() => {
   174→          setBillFor(null);
   175→          load();
   176→        }}
   177→      />
   178→    </div>
   179→  );
   180→}

   186→  onBill,
   187→  onToggleMaintenance,

   217→

   219→        <div className="mt-4 flex-1 space-y-1.5 rounded-xl border border-white/5 bg-white/[0.03] p-3">

   221→          <p className="text-xs text-muted-foreground">{station.activeSession.customerPhone} · from {fmtTime(station.activeSession.startedAt)}</p>

   274→function StartSessionDialog({
   275→  station,
   276→  onClose,
   277→  onStarted,
   278→}: {
   279→  station: FloorStation | null;
   280→  onClose: () => void;
   281→  onStarted: () => void;
   282→}) {
   283→  const [tab, setTab] = React.useState<"existing" | "new">("existing");
   284→  const [q, setQ] = React.useState("");
   285→  const [results, setResults] = React.useState<CustomerLite[]>([]);
   286→  const [selected, setSelected] = React.useState<CustomerLite | null>(null);
   287→  const [name, setName] = React.useState("");
   288→  const [phone, setPhone] = React.useState("");
   289→  const [planned, setPlanned] = React.useState("");
   290→  const [busy, setBusy] = React.useState(false);
   291→
   292→  React.useEffect(() => {
   293→    if (!station) return;
   294→    setTab("existing");
   295→    setQ("");
   296→    setResults([]);
   297→    setSelected(null);
   298→    setName("");
   299→    setPhone("");
   300→    setPlanned("");
   301→  }, [station]);
   302→
   303→  React.useEffect(() => {
   304→    if (tab !== "existing" || !station) return;
   305→    const t = setTimeout(async () => {
   306→      try {
   307→        const res = await fetch(`/api/members?q=${encodeURIComponent(q)}`, { cache: "no-store" });
   308→        if (!res.ok) return;
   309→        const json = await res.json();
   310→        setResults(json.customers ?? []);
   311→      } catch {
   312→        /* ignore */
   313→      }
   314→    }, 250);
   315→    return () => clearTimeout(t);
   316→  }, [q, tab, station]);
   317→
   318→  async function start() {
   319→    if (!station) return;
   320→    setBusy(true);
   321→    try {
   322→      const body: Record<string, unknown> = { stationId: station.id };
   323→      if (planned && Number(planned) > 0) body.plannedMinutes = Number(planned);
   324→      if (tab === "existing" && selected) body.customerId = selected.id;
   325→      else if (tab === "new") body.quickCustomer = { name, phone };
   326→
   327→      const res = await fetch("/api/sessions", {
   328→        method: "POST",
   329→        headers: { "Content-Type": "application/json" },
   330→        body: JSON.stringify(body),
   331→      });
   332→      const json = await res.json().catch(() => ({}));
   333→      if (!res.ok) throw new Error(json.error || "Could not start session");
   334→      toast.success(`${station.name} started for ${(json.session?.customer?.name as string) || "guest"}`, {
   335→        description: json.rate?.ruleName ? `Rate ${formatINR(json.rate.rate)}/hr — ${json.rate.ruleName}` : `Rate ${formatINR(json.rate?.rate ?? station.hourlyRate)}/hr`,
   336→      });
   337→      onStarted();
   338→    } catch (e) {
   339→      toast.error(e instanceof Error ? e.message : "Could not start session");
   340→    } finally {
   341→      setBusy(false);
   342→    }
   343→  }
   344→
   345→  const canStart = tab === "existing" ? !!selected : name.trim().length > 1 && phone.replace(/\D/g, "").length >= 10;
   346→
   347→  return (
   348→    <Dialog open={!!station} onOpenChange={(v) => !v && onClose()}>
   349→      <DialogContent className="max-h-[85vh] overflow-y-auto border-violet-500/25 bg-[#0f1526] scroll-neon sm:max-w-md">
   350→        <DialogHeader>
   351→          <DialogTitle className="font-display">Start Session — {station?.name}</DialogTitle>
   352→          <DialogDescription>
   353→            {station ? `${TYPE_STYLES[station.type].label} · ${formatINR(station.hourlyRate)}/hr base rate` : ""}
   354→          </DialogDescription>
   355→        </DialogHeader>
   356→
   357→        <div className="grid grid-cols-2 gap-2">
   358→          <Button variant={tab === "existing" ? "default" : "outline"} size="sm" onClick={() => setTab("existing")}
   359→            className={tab === "existing" ? "bg-violet-600" : "border-violet-500/30"}>
   360→            <Users className="mr-1.5 h-4 w-4" /> Existing
   361→          </Button>
   362→          <Button variant={tab === "new" ? "default" : "outline"} size="sm" onClick={() => setTab("new")}
   363→            className={tab === "new" ? "bg-violet-600" : "border-violet-500/30"}>
   364→            <UserPlus className="mr-1.5 h-4 w-4" /> Quick Add
   365→          </Button>
   366→        </div>
   367→
   368→        {tab === "existing" ? (
   369→          <div className="space-y-2">
   370→            <div className="relative">
   371→              <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" aria-hidden />
   372→              <Input value={q} onChange={(e) => setQ(e.target.value)} placeholder="Search phone or name…" className="border-violet-500/25 bg-white/5 pl-9" />
   373→            </div>
   374→            <div className="max-h-52 space-y-1.5 overflow-y-auto scroll-neon pr-1">
   375→              {results.length === 0 ? (
   376→                <p className="py-6 text-center text-xs text-muted-foreground">No members found — use Quick Add.</p>
   377→              ) : (
   378→                results.map((c) => (
   379→                  <button
   380→                    key={c.id}
   381→                    type="button"
   382→                    onClick={() => setSelected(c)}
   383→                    className={
   384→                      "flex w-full items-center justify-between rounded-xl border px-3 py-2.5 text-left transition-colors " +
   385→                      (selected?.id === c.id
   386→                        ? "border-violet-500/60 bg-violet-500/15"
   387→                        : "border-white/10 bg-white/[0.03] hover:border-violet-500/30")
   388→                    }
   389→                  >
   390→                    <span>
   391→                      <span className="block text-sm font-medium">{c.name}</span>
   392→                      <span className="block text-xs text-muted-foreground">{c.phone}</span>
   393→                    </span>
   394→                    <span className="text-right">
   395→                      <span className="block text-xs text-amber-300">{formatINR(c.walletBalance)}</span>
   396→                      <span className="block text-[10px] text-violet-300">{c.loyaltyPoints} pts</span>
   397→                    </span>
   398→                  </button>
   399→                ))
   400→              )}
   401→            </div>
   402→          </div>
   403→        ) : (
   404→          <div className="space-y-3">
   405→            <div className="space-y-1.5">
   406→              <Label htmlFor="qc-name">Name</Label>
   407→              <Input id="qc-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="Customer name" className="border-violet-500/25 bg-white/5" />
   408→            </div>
   409→            <div className="space-y-1.5">
   410→              <Label htmlFor="qc-phone">Phone</Label>
   411→              <Input id="qc-phone" value={phone} onChange={(e) => setPhone(e.target.value)} placeholder="98765 43210" inputMode="tel" className="border-violet-500/25 bg-white/5" />
   412→            </div>
   413→          </div>
   414→        )}
   415→
   416→        <div className="space-y-1.5">
   417→          <Label htmlFor="qc-planned">Planned minutes (optional)</Label>
   418→          <Input id="qc-planned" value={planned} onChange={(e) => setPlanned(e.target.value.replace(/\D/g, ""))} placeholder="e.g. 60" inputMode="numeric" className="border-violet-500/25 bg-white/5" />
   419→        </div>
   420→
   421→        <Button
   422→          onClick={start}
   423→          disabled={!canStart || busy}
   424→          className="h-11 bg-gradient-to-r from-violet-600 to-cyan-500 font-semibold text-white hover:opacity-90"
   425→        >
   426→          {busy ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Zap className="mr-2 h-4 w-4" />}
   427→          {busy ? "Starting…" : "Start Session"}
   428→        </Button>
   429→      </DialogContent>
   430→    </Dialog>
   431→  );
   432→}

   434→function BillDialog({
   435→  station,
   436→  onClose,
   437→  onEnded,
   438→}: {
   439→  station: FloorStation | null;
   440→  onClose: () => void;
   441→  onEnded: () => void;
   442→}) {
   443→  const [detail, setDetail] = React.useState<SessionDetail | null>(null);
   444→  const [discount, setDiscount] = React.useState("");
   445→  const [pay, setPay] = React.useState({ UPI: "", CASH: "", CARD: "" });
   446→  const [useWallet, setUseWallet] = React.useState(false);
   447→  const [walletAmount, setWalletAmount] = React.useState("");
   448→  const [busy, setBusy] = React.useState(false);
   449→  const [now, setNow] = React.useState(() => Date.now());
   450→
   451→  const load = React.useCallback(async () => {
   452→    if (!station?.activeSession) return;
   453→    try {
   454→      const res = await fetch(`/api/sessions/${station.activeSession.id}`, { cache: "no-store" });
   455→      if (!res.ok) throw new Error();
   456→      const json: SessionDetail = await res.json();
   457→      setDetail(json);
   458→    } catch {
   459→      toast.error("Could not load session bill");
   460→    }
   461→  }, [station]);
   462→
   463→  React.useEffect(() => {
   464→    if (!station) return;
   465→    setDetail(null);
   466→    setDiscount("");
   467→    setPay({ UPI: "", CASH: "", CARD: "" });
   468→    setUseWallet(false);
   469→    setWalletAmount("");
   470→    load();
   471→    const t = setInterval(() => setNow(Date.now()), 1000);
   472→    return () => clearInterval(t);
   473→  }, [station, load]);
   474→
   475→  const liveMinutes = detail
   476→    ? Math.max(1, Math.ceil(((detail.session.endedAt ? new Date(detail.session.endedAt).getTime() : now) - new Date(detail.session.startedAt).getTime()) / 60000))
   477→    : 0;
   478→  const liveFee = detail ? Math.round((detail.session.rateSnapshot * liveMinutes) / 60) : 0;
   479→  const fnbTotal = detail ? detail.session.orders.reduce((s, o) => s + o.total, 0) : 0;
   480→  const discountNum = Math.min(Number(discount) || 0, liveFee + fnbTotal);
   481→  const due = Math.max(0, Math.round(liveFee + fnbTotal - discountNum));
   482→  const walletNum = useWallet ? Math.min(Number(walletAmount) || 0, detail?.session.customer?.walletBalance ?? 0) : 0;
   483→  const walletExcess = useWallet && (Number(walletAmount) || 0) > (detail?.session.customer?.walletBalance ?? 0);
   484→  const allocated = (Number(pay.UPI) || 0) + (Number(pay.CASH) || 0) + (Number(pay.CARD) || 0) + walletNum;
   485→  const remaining = due - allocated;
   486→  const canConfirm = detail && remaining === 0 && !walletExcess && due >= 0;
   487→
   488→  async function confirmEnd() {
   489→    if (!detail) return;
   490→    setBusy(true);
   491→    try {
   492→      const payments = (["UPI", "CASH", "CARD"] as const)
   493→        .map((m) => ({ method: m, amount: Number(pay[m]) || 0 }))
   494→        .filter((p) => p.amount > 0);
   495→      const res = await fetch("/api/sessions/end", {
   496→        method: "POST",
   497→        headers: { "Content-Type": "application/json" },
   498→        body: JSON.stringify({
   499→          sessionId: detail.session.id,
   500→          discount: discountNum,
   501→          payments,
   502→          walletAmount: walletNum,
   503→          walletCustomerId: detail.session.customer?.id,
   504→        }),
   505→      });
   506→      const json = await res.json().catch(() => ({}));
   507→      if (!res.ok) throw new Error(json.error || "Billing failed");
   508→      toast.success(`Billed ${formatINR(json.bill?.total ?? due)} — ${json.bill?.loyaltyEarned ?? 0} loyalty points earned`, {
   509→        description: `${detail.session.station.name} is now free`,
   510→      });
   511→      onEnded();
   512→    } catch (e) {
   513→      toast.error(e instanceof Error ? e.message : "Billing failed");
   514→    } finally {
   515→      setBusy(false);
   516→    }
   517→  }
   518→
   519→  return (
   520→    <Dialog open={!!station} onOpenChange={(v) => !v && onClose()}>
   521→      <DialogContent className="max-h-[88vh] overflow-y-auto border-violet-500/25 bg-[#0f1526] scroll-neon sm:max-w-lg">
   522→        <DialogHeader>
   523→          <DialogTitle className="font-display">End Session & Bill — {detail?.session.station.name ?? station?.name}</DialogTitle>
   524→          <DialogDescription>Started {detail ? fmtDateTime(detail.session.startedAt) : "…"}</DialogDescription>
   525→        </DialogHeader>
   526→
   527→        {!detail ? (
   528→          <div className="space-y-3 py-4">
   529→            <Skeleton className="h-16 bg-white/5" />
   530→            <Skeleton className="h-16 bg-white/5" />
   531→          </div>
   532→        ) : (
   533→          <div className="space-y-4">
   534→            {/* Line items */}
   535→            <div className="rounded-xl border border-white/10 bg-white/[0.03] p-4">
   536→              <p className="mb-2 text-xs font-semibold uppercase tracking-wider text-muted-foreground">Bill lines</p>
   537→              <div className="flex items-center justify-between text-sm">
   538→                <span>
   539→                  {detail.session.station.name} · {liveMinutes} min × {formatINR(detail.session.rateSnapshot)}/hr
   540→                </span>
   541→                <span className="tabular font-semibold">{formatINR(liveFee)}</span>
   542→              </div>
   543→              {detail.session.orders.flatMap((o) =>
   544→                o.items.map((it, i) => (
   545→                  <div key={o.id + "-" + i} className="mt-1.5 flex items-center justify-between text-sm text-muted-foreground">
   546→                    <span>🍿 {it.name} × {it.qty}</span>
   547→                    <span className="tabular">{formatINR(it.price * it.qty)}</span>
   548→                  </div>
   549→                ))
   550→              )}
   551→              {fnbTotal > 0 ? (
   552→                <div className="mt-1.5 flex items-center justify-between text-sm">
   553→                  <span className="font-medium">F&B total</span>
   554→                  <span className="tabular font-semibold">{formatINR(fnbTotal)}</span>
   555→                </div>
   556→              ) : (
   557→                <p className="mt-1.5 text-xs text-muted-foreground">No F&B on this bill</p>
   558→              )}
   559→              <Separator className="my-2.5 bg-white/10" />
   560→              <div className="flex items-center justify-between text-sm text-muted-foreground">
   561→                <span>Discount</span>
   562→                <span className="tabular text-emerald-300">− {formatINR(discountNum)}</span>
   563→              </div>
   564→              <div className="mt-1 flex items-baseline justify-between">
   565→                <span className="font-display text-sm font-semibold">TOTAL DUE</span>
   566→                <span className="font-display text-2xl font-extrabold neon-text tabular">{formatINR(due)}</span>
   567→              </div>
   568→            </div>
   569→
   570→            {/* Discount */}
   571→            <div className="space-y-1.5">
   572→              <Label htmlFor="bill-discount">Discount (₹)</Label>
   573→              <div className="flex gap-2">
   574→                <Input id="bill-discount" value={discount} onChange={(e) => setDiscount(e.target.value.replace(/[^\d.]/g, ""))} placeholder="0" inputMode="decimal" className="border-violet-500/25 bg-white/5" />
   575→                {[0, 20, 50].map((d) => (
   576→                  <Button key={d} type="button" variant="outline" size="sm" className="border-violet-500/30" onClick={() => setDiscount(String(d))}>
   577→                    ₹{d}
   578→                  </Button>
   579→                ))}
   580→              </div>
   581→            </div>
   582→
   583→            {/* Payments */}
   584→            <div className="space-y-2.5 rounded-xl border border-white/10 bg-white/[0.03] p-4">
   585→              <p className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">Payment — split allowed</p>
   586→              {(["UPI", "CASH", "CARD"] as const).map((m) => (
   587→                <div key={m} className="flex items-center gap-3">
   588→                  <Label htmlFor={"pay-" + m} className="w-14 text-sm">{m}</Label>
   589→                  <Input
   590→                    id={"pay-" + m}
   591→                    value={pay[m]}
   592→                    onChange={(e) => setPay((p) => ({ ...p, [m]: e.target.value.replace(/[^\d.]/g, "") }))}
   593→                    placeholder="0"
   594→                    inputMode="decimal"
   595→                    className="border-violet-500/25 bg-white/5"
   596→                  />
   597→                </div>
   598→              ))}
   599→              {detail.session.customer ? (
   600→                <div className="rounded-lg border border-amber-500/25 bg-amber-500/5 p-3">
   601→                  <div className="flex items-center justify-between">
   602→                    <Label className="text-sm text-amber-200">Use wallet ({formatINR(detail.session.customer.walletBalance)} available)</Label>
   603→                    <input
   604→                      type="checkbox"
   605→                      checked={useWallet}
   606→                      onChange={(e) => {
   607→                        setUseWallet(e.target.checked);
   608→                        if (e.target.checked) setWalletAmount(String(Math.min(due, Math.round(detail.session.customer!.walletBalance))));
   609→                        else setWalletAmount("");
   610→                      }}
   611→                      className="h-4 w-4 accent-violet-500"
   612→                      aria-label="Debit wallet"
   613→                    />
   614→                  </div>
   615→                  {useWallet ? (
   616→                    <div className="mt-2 space-y-1.5">
   617→                      <Input
   618→                        value={walletAmount}
   619→                        onChange={(e) => setWalletAmount(e.target.value.replace(/[^\d.]/g, ""))}
   620→                        placeholder="Wallet debit amount"
   621→                        inputMode="decimal"
   622→                        className="border-amber-500/30 bg-white/5"
   623→                      />
   624→                      {walletExcess ? (
   625→                        <p className="text-xs font-medium text-red-400">Insufficient balance — {formatINR(detail.session.customer.walletBalance)} available</p>
   626→                      ) : null}
   627→                    </div>
   628→                  ) : null}
   629→                </div>
   630→              ) : null}
   631→
   632→              <div className="flex items-center justify-between pt-1">
   633→                <span className="text-xs text-muted-foreground">Remaining to allocate</span>
   634→                <Button
   635→                  type="button"
   636→                  variant="ghost"
   637→                  size="sm"
   638→                  className="h-7 text-xs text-cyan-300"
   639→                  onClick={() => {
   640→                    if (remaining > 0) {
   641→                      setPay((p) => ({ ...p, UPI: String((Number(p.UPI) || 0) + remaining) }));
   642→                    }
   643→                  }}
   644→                  disabled={remaining <= 0}
   645→                >
   646→                  + settle via UPI
   647→                </Button>
   648→              </div>
   649→              <p className={"text-sm font-semibold tabular " + (remaining === 0 ? "text-emerald-400" : remaining > 0 ? "text-amber-400" : "text-red-400")}>
   650→                {formatINR(remaining)} {remaining === 0 ? "— balanced ✓" : remaining > 0 ? "still due" : "overpaid"}
   651→              </p>
   652→            </div>
   653→
   654→            <Button
   655→              onClick={confirmEnd}
   656→              disabled={!canConfirm || busy}
   657→              className="h-12 w-full bg-gradient-to-r from-violet-600 to-cyan-500 font-display text-sm font-bold tracking-wide text-white hover:opacity-90"
   658→            >
   659→              {busy ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Plus className="mr-2 h-4 w-4" />}
   660→              {busy ? "Closing…" : `Confirm & Bill ${formatINR(due)}`}
   661→            </Button>
   662→          </div>
   663→        )}
   664→      </DialogContent>
   665→    </Dialog>
   666→  );
   667→}
