   404→   404→  onCreated: () => void;
   405→   405→}) {
   406→   406→  const [stationId, setStationId] = React.useState(initialStation?.id ?? "");
   407→   407→  const [hour, setHour] = React.useState(String(initialHour ?? new Date().getHours() + 1));

   409→   409→  const [q, setQ] = React.useState("");
   410→   410→  const [results, setResults] = React.useState<CustomerLite[]>([]);

   502→   502→
   503→   503→        <div className="space-y-4">

   514→   514→                  <SelectItem key={s.id} value={s.id}>
   515→   515→                    {s.name} · {s.type} · ₹{s.hourlyRate}/hr
   516→   516→                  </SelectItem>
   517→   517→                ))}
   518→   518→              </SelectContent>
   519→   519→            </Select>

   521→   521→
   522→   522→          <div className="grid grid-cols-3 gap-2">
   523→   523→            <div className="col-span-1 space-y-1.5">

   527→   527→                <SelectContent className="max-h-56 border-violet-500/25 bg-[#0f1526]">
   528→   528→                  {HOURS.map((h) => (
   529→   529→                    <SelectItem key={h} value={String(h)}>{hourLabel(h)}</SelectItem>
   530→   530→                  ))}
   531→   531→                </SelectContent>
   532→   532→              </Select>
   533→   533→            </div>
   534→   534→            <div className="col-span-1 space-y-1.5">
   535→   535→              <Label>Duration</Label>
   536→   536→              <Select value={duration} onValueChange={setDuration}>
   537→   537→                <SelectTrigger className="border-violet-500/25 bg-white/5"><SelectValue /></SelectTrigger>
   538→   538→                <SelectContent className="border-violet-500/25 bg-[#0f1526]">
