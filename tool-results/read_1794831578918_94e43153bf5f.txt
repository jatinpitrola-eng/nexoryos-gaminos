    72→export default function BookingsPage() {
    73→  const [date, setDate] = React.useState(() => new Date().toISOString().slice(0, 10));
    74→  const [bookings, setBookings] = React.useState<Booking[] | null>(null);
    75→  const [stations, setStations] = React.useState<StationLite[]>([]);
    76→  const [createFor, setCreateFor] = React.useState<{ station?: StationLite; hour?: number } | null>(null);
    77→  const [actionBusy, setActionBusy] = React.useState<string | null>(null);
    78→
    79→  const load = React.useCallback(async () => {
    80→    try {
    81→      const res = await fetch(`/api/bookings?date=${date}`, { cache: "no-store" });
    82→      if (!res.ok) throw new Error();
    83→      const json = await res.json();
    84→      setBookings(json.bookings);
    85→    } catch {
    86→      toast.error("Could not load bookings");
    87→    }
    88→  }, [date]);
    89→
    90→  React.useEffect(() => {
    91→    load();
    92→    const t = setInterval(load, 30000);
    93→    return () => clearInterval(t);
    94→  }, [load]);
    95→
    96→  React.useEffect(() => {
    97→    fetch("/api/stations", { cache: "no-store" })
    98→      .then((r) => (r.ok ? r.json() : { stations: [] }))
    99→      .then((j) => setStations(j.stations ?? []))
   100→      .catch(() => {});
   101→  }, []);
   102→
   103→  const dayStart = React.useMemo(() => startOfDay(new Date(date + "T00:00:00")), [date]);
   104→  const now = new Date();
   105→  const isToday = isSameDay(dayStart, now);
   106→
   107→  const upcoming = React.useMemo(
   108→    () =>
   109→      (bookings ?? [])
   110→        .filter((b) => b.status === "BOOKED")
   111→        .sort((a, b) => a.slotStart.localeCompare(b.slotStart)),
   112→    [bookings]
   113→  );
   114→
   115→  function shiftDate(days: number) {
   116→    const d = new Date(date + "T00:00:00");
   117→    d.setDate(d.getDate() + days);
   118→    setDate(d.toISOString().slice(0, 10));
   119→  }
   120→
   121→  async function bookingAction(b: Booking, action: "CHECKIN" | "NOSHOW" | "CANCEL") {
   122→    if (action === "CANCEL" && !window.confirm(`Cancel booking ${b.code} for ${b.customerName}?`)) return;
   123→    setActionBusy(b.id);
   124→    try {
   125→      const res = await fetch(`/api/bookings/${b.id}`, {
   126→        method: "PATCH",
   127→        headers: { "Content-Type": "application/json" },
   128→        body: JSON.stringify({ action }),
   129→      });
   130→      const json = await res.json().catch(() => ({}));
   131→      if (!res.ok) throw new Error(json.error || "Action failed");
   132→      if (action === "CHECKIN")
   133→        toast.success(`${b.stationName} — session started for ${b.customerName}`);
   134→      else if (action === "NOSHOW") toast.success(`${b.code} marked no-show`);
   135→      else toast.success(`${b.code} cancelled`);
   136→      await load();
   137→    } catch (e) {
   138→      toast.error(e instanceof Error ? e.message : "Action failed");
   139→    } finally {
   140→      setActionBusy(null);
   141→    }
   142→  }
   143→
   144→  function reminderText(b: Booking): string {
   145→    return [
   146→      `Hi ${b.customerName}! 👋 Your slot at our gaming zone is confirmed:`,
   147→      `🎮 ${b.stationName} (${TYPE_STYLES[b.stationType as keyof typeof TYPE_STYLES]?.label ?? b.stationType})`,
   148→      `📅 ${fmtDate(b.slotStart)} · ${fmtTime(b.slotStart)} – ${fmtTime(b.slotEnd)}`,
   149→      `💰 Deposit: ${formatINR(b.depositAmount)} (${b.depositPaid ? "received" : "pending at counter"})`,
   150→      `🔑 Booking code: ${b.code}`,
   151→      `See you soon — reply CALL if you need to change your slot!`,
   152→    ].join("\n");
   153→  }
   154→
   155→  function copy(text: string, label: string) {

   157→      .writeText(text)
   158→      .then(() => toast.success(`${label} copied`))
   159→      .catch(() => toast.error("Copy failed"));
   160→  }
   161→
   162→  return (
   163→    <div className="mx-auto max-w-7xl space-y-6">
   164→      <header className="flex flex-wrap items-center justify-between gap-3">
   165→        <div>
   166→          <h1 className="font-display text-xl font-bold sm:text-2xl">Bookings</h1>
   167→          <p className="text-sm text-muted-foreground">Hourly slots, deposits & check-ins</p>
   168→        </div>
   169→        <div className="flex items-center gap-2">
   170→          <Button variant="outline" size="icon" className="border-violet-500/30" onClick={() => shiftDate(-1)} aria-label="Previous day">
   171→            <ChevronLeft className="h-4 w-4" />
   172→          </Button>
   173→          <Input
   174→            type="date"
   175→            value={date}
   176→            onChange={(e) => setDate(e.target.value)}
   177→            className="w-40 border-violet-500/25 bg-white/5"
   178→            aria-label="Booking date"
   179→          />
   180→          <Button variant="outline" size="icon" className="border-violet-500/30" onClick={() => shiftDate(1)} aria-label="Next day">
   181→            <ChevronRight className="h-4 w-4" />
   182→          </Button>
   183→          <Button variant="outline" size="sm" className="border-cyan-500/40 text-cyan-300" onClick={() => setDate(new Date().toISOString().slice(0, 10))}>
   184→            Today
   185→          </Button>
   186→          <Button className="bg-gradient-to-r from-violet-600 to-cyan-500 text-white hover:opacity-90" onClick={() => setCreateFor({})}>
   187→            <Plus className="mr-1.5 h-4 w-4" /> New Booking
   188→          </Button>
   189→        </div>
   190→      </header>
   191→
   192→      {/* Upcoming today */}

   194→        <h2 className="font-display text-base font-semibold tracking-wide">UPCOMING {isToday ? "TODAY" : fmtDate(date).toUpperCase()}</h2>
   195→        {!bookings ? (
   196→          <Skeleton className="h-24 rounded-2xl bg-white/5" />
   197→        ) : upcoming.length === 0 ? (

   201→        ) : (
   202→          <div className="grid gap-3 lg:grid-cols-2">
   203→            {upcoming.map((b) => {
   204→              const start = new Date(b.slotStart);
   205→              const isUpcoming = start.getTime() >= now.getTime();
   206→              return (
   207→                <GlassCard key={b.id} className={"p-4 " + (isUpcoming ? "" : "opacity-60")}>
   208→                  <div className="flex flex-wrap items-start justify-between gap-3">
   209→                    <div className="min-w-0">
   210→                      <p className="font-display text-sm font-bold tabular">
   211→                        {fmtTime(b.slotStart)} – {fmtTime(b.slotEnd)}
   212→                      </p>
   213→                      <p className="mt-1 flex flex-wrap items-center gap-2 text-sm">
   214→                        <span className={"inline-flex rounded-full border px-2 py-0.5 text-[10px] font-semibold uppercase " + TYPE_STYLES[b.stationType as keyof typeof TYPE_STYLES]?.badge}>
   215→                          {b.stationName}
   216→                        </span>
   217→                        <span className="font-medium">{b.customerName}</span>
   218→                        <span className="text-xs text-muted-foreground">{b.customerPhone}</span>
   219→                      </p>
   220→                      <p className="mt-1.5 flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   221→                        <Badge variant="outline" className="border-white/15 text-[10px]">{b.source}</Badge>
   222→                        <span>
   223→                          Deposit {formatINR(b.depositAmount)}{" "}
   224→                          {b.depositPaid ? (
   225→                            <span className="font-semibold text-emerald-400">Paid</span>
   226→                          ) : (
   227→                            <span className="font-semibold text-amber-400">Unpaid</span>
   228→                          )}
   229→                        </span>
   230→                      </p>
   231→                    </div>
   232→                    <div className="flex flex-col items-end gap-2">
   233→                      <button
   234→                        type="button"
   235→                        onClick={() => copy(b.code, "Booking code")}
   236→                        className="flex items-center gap-1 rounded-lg border border-violet-500/30 bg-violet-500/10 px-2 py-1 font-mono text-xs text-violet-200 hover:bg-violet-500/20"
   237→                        aria-label="Copy booking code"
   238→                      >
   239→                        {b.code} <Copy className="h-3 w-3" />
   240→                      </button>
   241→                      <button
   242→                        type="button"
   243→                        onClick={() => copy(reminderText(b), "Reminder text")}
   244→                        className="flex items-center gap-1 text-xs text-emerald-300 hover:text-emerald-200"
   245→                        aria-label="Copy WhatsApp reminder"
   246→                      >
   247→                        <MessageCircle className="h-3.5 w-3.5" /> Copy reminder
   248→                      </button>
   249→                    </div>
   250→                  </div>
   251→                  <div className="mt-3 flex flex-wrap gap-2 border-t border-white/5 pt-3">
   252→                    <Button size="sm" variant="outline" className="h-9 border-emerald-500/40 text-emerald-300 hover:bg-emerald-500/10"
   253→                      onClick={() => bookingAction(b, "CHECKIN")} disabled={actionBusy === b.id}>
   254→                      {actionBusy === b.id ? <Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> : <CheckCircle2 className="mr-1.5 h-3.5 w-3.5" />}
   255→                      Check-in
   256→                    </Button>
   257→                    <Button size="sm" variant="outline" className="h-9 border-amber-500/40 text-amber-300 hover:bg-amber-500/10"
   258→                      onClick={() => bookingAction(b, "NOSHOW")} disabled={actionBusy === b.id}>
   259→                      <UserX className="mr-1.5 h-3.5 w-3.5" /> No-show
   260→                    </Button>
   261→                    <Button size="sm" variant="outline" className="h-9 border-red-500/40 text-red-300 hover:bg-red-500/10"
   262→                      onClick={() => bookingAction(b, "CANCEL")} disabled={actionBusy === b.id}>
   263→                      <XCircle className="mr-1.5 h-3.5 w-3.5" /> Cancel
   264→                    </Button>
   265→                  </div>
   266→                </GlassCard>
   267→              );
   268→            })}
   269→          </div>
   270→        )}
   271→      </section>
   272→
   273→      {/* Day calendar */}
   274→      <section aria-label="Day calendar" className="space-y-3">
   275→        <div className="flex flex-wrap items-center justify-between gap-2">
   276→          <h2 className="font-display text-base font-semibold tracking-wide">DAY CALENDAR</h2>
   277→          <div className="flex items-center gap-3 text-[11px] text-muted-foreground">
   278→            <span className="flex items-center gap-1.5"><StatusDot state="DEPOSIT" pulse={false} /> Deposit paid</span>
   279→            <span className="flex items-center gap-1.5"><StatusDot state="BOOKED" pulse={false} /> Booked</span>
   280→            <span className="flex items-center gap-1.5"><StatusDot state="ACTIVE" pulse={false} /> Checked-in</span>
   281→          </div>
   282→        </div>
   283→        <p className="text-xs text-muted-foreground sm:hidden">Scroll sideways to see all hours →</p>
   284→        <div className="overflow-x-auto rounded-2xl border border-violet-500/15 bg-[#0c1122]/60 scroll-neon">
   285→          <div className="min-w-[900px]">
   286→            {/* Hour header */}
   287→            <div className="grid" style={{ gridTemplateColumns: `130px repeat(${HOURS.length}, minmax(76px, 1fr))` }}>
   288→              <div className="border-b border-r border-white/5 p-2.5 text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   289→                Station
   290→              </div>
   291→              {HOURS.map((h) => {
   292→                const cellStart = new Date(dayStart);
   293→                cellStart.setHours(h, 0, 0, 0);
   294→                const isNow = isToday && now.getHours() === h;
   295→                return (
   296→                  <div
   297→                    key={h}
   298→                    className={
   299→                      "border-b border-white/5 p-1.5 text-center text-[10px] font-medium " +
   300→                      (isNow ? "border-t-2 border-t-cyan-400 bg-cyan-500/5 text-cyan-300" : "text-muted-foreground")
   301→                    }
   302→                  >
   303→                    {hourLabel(h)}
   304→                  </div>
   305→                );
   306→              })}
   307→            </div>
   308→            {/* Station rows */}
   309→            {(stations.length ? stations : []).map((st) => {
   310→              const maint = st.status === "MAINTENANCE";
   311→              return (
   312→                <div key={st.id} className="grid" style={{ gridTemplateColumns: `130px repeat(${HOURS.length}, minmax(76px, 1fr))` }}>
   313→                  <div className={"border-b border-r border-white/5 p-2.5 " + (maint ? "opacity-50" : "")}>
   314→                    <p className="truncate text-xs font-semibold">{st.name}</p>
   315→                    <p className="text-[10px] text-muted-foreground">{TYPE_STYLES[st.type as keyof typeof TYPE_STYLES]?.label ?? st.type}</p>
   316→                  </div>
   317→                  {HOURS.map((h) => {
   318→                    const cellStart = new Date(dayStart);
   319→                    cellStart.setHours(h, 0, 0, 0);
   320→                    const cellEnd = new Date(cellStart.getTime() + 3600_000);
   321→                    const covered = (bookings ?? []).filter(
   322→                      (b) =>
   323→                        b.stationId === st.id &&
   324→                        ["BOOKED", "CHECKEDIN"].includes(b.status) &&
   325→                        minutesOverlap(cellStart, cellEnd, new Date(b.slotStart), new Date(b.slotEnd)) > 0
   326→                    );
   327→                    const isPast = isToday && cellEnd.getTime() < now.getTime();
   328→                    const b = covered[0];
   329→
   330→                    if (b) {
   331→                      const bg =
   332→                        b.status === "CHECKEDIN"
   333→                          ? "bg-violet-500/25 border-violet-500/50"
   334→                          : b.depositPaid
   335→                            ? "bg-emerald-500/20 border-emerald-500/40"
   336→                            : "bg-amber-500/20 border-amber-500/40";
   337→                      return (
   338→                        <div key={h} className={"m-1 rounded-md border p-1 text-[9px] leading-tight " + bg} title={`${b.code} · ${b.customerName}`}>
   339→                          <p className="truncate font-semibold">{b.customerName.split(" ")[0]}</p>
   340→                          <p className="truncate font-mono opacity-80">{b.code}</p>
   341→                        </div>
   342→                      );
   343→                    }
   344→
   345→                    return (
   346→                      <button
   347→                        key={h}
   348→                        type="button"
   349→                        disabled={isPast || maint}
   350→                        onClick={() => setCreateFor({ station: st, hour: h })}
   351→                        className={
   352→                          "m-1 rounded-md border p-1 text-[9px] transition-colors " +
   353→                          (isPast || maint
   354→                            ? "border-white/5 bg-white/[0.02] opacity-40"
   355→                            : "border-white/5 bg-white/[0.03] hover:border-violet-500/40 hover:bg-violet-500/10")
   356→                        }
   357→                        aria-label={maint ? `${st.name} under maintenance` : `Book ${st.name} ${hourLabel(h)}`}
   358→                      >
   359→                        {maint ? "🛠" : isPast ? "" : "+"}
   360→                      </button>
   361→                    );
   362→                  })}
   363→                </div>
   364→              );
   365→            })}
   366→          </div>
   367→        </div>
   368→      </section>
   369→
   370→      <CreateBookingDialog
   371→        open={!!createFor}
   372→        stations={stations}
   373→        initialStation={createFor?.station}
   374→        initialHour={createFor?.hour}
   375→        date={date}
   376→        bookings={bookings ?? []}
   377→        onClose={() => setCreateFor(null)}
   378→        onCreated={() => {
   379→          setCreateFor(null);
   380→          load();
   381→        }}
   382→      />
   383→    </div>
   384→  );
   385→}
   386→
   387→function CreateBookingDialog({
   388→  open,
   389→  stations,
   390→  initialStation,
   391→  initialHour,
   392→  date,
   393→  bookings,
   394→  onClose,
   395→  onCreated,
   396→}: {
   397→  open: boolean;
   398→  stations: StationLite[];
   399→  initialStation?: StationLite;
   400→  initialHour?: number;
   401→  date: string;
   402→  bookings: Booking[];
   403→  onClose: () => void;
   404→  onCreated: () => void;
   405→}) {
   406→  const [stationId, setStationId] = React.useState(initialStation?.id ?? "");
   407→  const [hour, setHour] = React.useState(String(initialHour ?? new Date().getHours() + 1));
   408→  const [duration, setDuration] = React.useState("1");
   409→  const [q, setQ] = React.useState("");
   410→  const [results, setResults] = React.useState<CustomerLite[]>([]);
   411→  const [customer, setCustomer] = React.useState<CustomerLite | null>(null);
   412→  const [deposit, setDeposit] = React.useState("");
   413→  const [depositPaid, setDepositPaid] = React.useState(false);
   414→  const [source, setSource] = React.useState("PHONE");
   415→  const [busy, setBusy] = React.useState(false);
   416→
   417→  React.useEffect(() => {
   418→    if (!open) return;
   419→    setStationId(initialStation?.id ?? "");
   420→    setHour(String(initialHour ?? Math.min(21, Math.max(OPEN_HOUR, new Date().getHours() + 1))));
   421→    setDuration("1");
   422→    setQ("");
   423→    setResults([]);
   424→    setCustomer(null);
   425→    setDeposit(initialStation ? String(Math.round(initialStation.hourlyRate / 2)) : "");
   426→    setDepositPaid(false);
   427→    setSource("PHONE");
   428→  }, [open, initialStation, initialHour]);
   429→
   430→  React.useEffect(() => {
   431→    if (!open) return;
   432→    const t = setTimeout(async () => {
   433→      try {
   434→        const res = await fetch(`/api/members?q=${encodeURIComponent(q)}`, { cache: "no-store" });
   435→        if (!res.ok) return;
   436→        const json = await res.json();
   437→        setResults(json.customers ?? []);
   438→      } catch {
   439→        /* ignore */
   440→      }
   441→    }, 250);
   442→    return () => clearTimeout(t);
   443→  }, [q, open]);
   444→
   445→  const startISO = React.useMemo(() => {
   446→    const d = new Date(date + "T00:00:00");
   447→    d.setHours(Number(hour) || OPEN_HOUR, 0, 0, 0);
   448→    return d;
   449→  }, [date, hour]);
   450→  const endISO = new Date(startISO.getTime() + Number(duration) * 3600_000);
   451→
   452→  const conflict = React.useMemo(() => {
   453→    if (!stationId) return null;
   454→    return (
   455→      bookings.find(
   456→        (b) =>
   457→          b.stationId === stationId &&
   458→          ["BOOKED", "CHECKEDIN"].includes(b.status) &&
   459→          minutesOverlap(startISO, endISO, new Date(b.slotStart), new Date(b.slotEnd)) > 0
   460→      ) ?? null
   461→    );
   462→  }, [bookings, stationId, startISO, endISO]);
   463→
   464→  const canSubmit = stationId && customer && !conflict;
   465→
   466→  async function submit() {
   467→    setBusy(true);
   468→    try {
   469→      const res = await fetch("/api/bookings", {
   470→        method: "POST",
   471→        headers: { "Content-Type": "application/json" },
   472→        body: JSON.stringify({
   473→          stationId,
   474→          customerId: customer?.id,
   475→          slotStart: startISO.toISOString(),
   476→          slotEnd: endISO.toISOString(),
   477→          depositAmount: Number(deposit) || 0,
   478→          depositPaid,
   479→          source,
   480→        }),
   481→      });
   482→      const json = await res.json().catch(() => ({}));
   483→      if (!res.ok) throw new Error(json.error || "Could not create booking");
   484→      toast.success(`Booking ${json.booking?.code} created`, {
   485→        description: `${json.booking?.station?.name} · ${fmtTime(startISO)} – ${fmtTime(endISO)}`,
   486→      });
   487→      onCreated();
   488→    } catch (e) {
   489→      toast.error(e instanceof Error ? e.message : "Could not create booking");
   490→    } finally {
   491→      setBusy(false);
   492→    }
   493→  }
   494→
   495→  return (
   496→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
